// Ví dụ metadata trong Product.java của reference.
@Entity
@Table(name = "products", schema = "learning_jpa")
public class Product {
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(name = "name", nullable = false, length = 120)
    private String name;
    protected Product() { } // JPA cần constructor no-arg public hoặc protected.
}
