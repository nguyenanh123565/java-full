// Đoạn mở rộng có chủ ý, KHÔNG có trong schema/reference gốc.
// Trong ProductDetail entity (cần @Entity, @Id, constructor no-arg):
@OneToOne(fetch = FetchType.LAZY)
@JoinColumn(name = "product_id", unique = true, nullable = false)
private Product product; // owning side

// Trong Product:
@OneToOne(mappedBy = "product", fetch = FetchType.LAZY)
private ProductDetail detail; // inverse side
